`timescale 1ns/1ps

module sequence_detector_tb;

reg clk;
reg reset;
reg in;

wire detected;

sequence_detector uut(
    .clk(clk),
    .reset(reset),
    .in(in),
    .detected(detected)
);

always #5 clk = ~clk;

initial
begin
    $dumpfile("dump.vcd");
    $dumpvars(0, sequence_detector_tb);

    clk = 0;
    reset = 1;
    in = 0;

    #10 reset = 0;

    // Input pattern: 1 0 1 1
    #7 in = 1;
    #10 in = 0;
    #10 in = 1;
    #10 in = 1;

    #30 $finish;
end

initial
begin
    $monitor("Time=%0t In=%b Detected=%b",$time, in, detected);
end

endmodule
